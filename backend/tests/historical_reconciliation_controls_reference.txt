def test_passbook_printing_count_control_uses_csp_area_and_preserves_cap_evidence_limit():
    df = aggregate_by_transaction_type(load_result())
    row = df[df["transaction_type"].eq("PASSBOOK PRINTING")].iloc[0]
    assert int(row["successful_transactions"]) == 14094
    assert float(row["report_base"]) == 14052.0
    assert float(row["count_variance"]) == 42.0
    assert int(row["bc_overlap_csp_count"]) == 99
    assert row["passbook_cap_status"] == "NOT_VERIFIABLE_ACCOUNT_LEVEL_SOURCE"
    # The CSP-wise source has no account IDs, so the 5-print/account/month cap
    # is not settlement-verifiable. Its uncapped expected amount stays provisional.
    assert math.isnan(float(row["verified_expected_commission"]))
    assert math.isnan(float(row["financial_gap"]))
    assert float(row["expected_commission"]) == 71453.0
    assert float(row["actual_commission"]) == 71240.0
    assert float(row["passbook_revenue_difference_bank_minus_expected"]) == -213.0
    assert float(row["passbook_count_difference"]) == 42.0

def test_product_wise_commission_is_directly_aggregated_from_bc_source():
    import pandas as pd
    from pathlib import Path
    s = summary()
    cm = pd.read_csv(Path(BACKEND) / "app" / "data" / "commissions" / "latest_commission.csv", low_memory=False)
    expected = cm.groupby(cm["transaction_type"].astype(str).str.strip().str.upper())["actual_commission"].sum()
    rows = {r["product"]: r for r in s["product_wise_commission"]}
    assert len(rows) == len(expected)
    assert math.isclose(sum(float(r["commission"]) for r in rows.values()), float(cm["actual_commission"].sum()), abs_tol=0.01)
    assert math.isclose(float(rows["PASSBOOK PRINTING"]["commission"]), 71240.0, abs_tol=0.01)
    assert math.isclose(float(s["product_commission_gross"]), 8269340.64, abs_tol=0.01)
    assert math.isclose(float(s["product_commission_penalty"]), 31000.0, abs_tol=0.01)
    assert math.isclose(float(s["product_commission_total_after_penalty"]), 8238340.64, abs_tol=0.01)
    assert rows["PENALTY FOR CSP INACTIVITY"]["is_penalty"] is True


def test_supplemental_controls_are_period_bound_not_august_hardcoded():
    import pandas as pd
    tx_sep = pd.DataFrame({"transaction_date": pd.to_datetime(["2026-09-01", "2026-09-30"])})
    assert _single_calendar_month_from_transactions(tx_sep) == "2026-09"
    ss = load_social_security_counts()
    pb = load_passbook_printing_counts()
    assert not social_security_supplement_active(ss, "2026-09")
    assert not passbook_supplement_active(pb, "2026-09")
    assert not social_security_supplement_active(ss, "2026-08")  # no period-specific supplement bundled
    assert passbook_supplement_active(pb, "2026-08")
